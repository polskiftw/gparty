cmake_minimum_required(VERSION 3.28)
project(gdupe_77k_benchmark LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(gdupe_77k_benchmark ../benchmark_77k.cpp)
if(MSVC)
  target_compile_options(gdupe_77k_benchmark PRIVATE /O2 /W4 /WX /EHsc /permissive- /utf-8)
else()
  target_compile_options(gdupe_77k_benchmark PRIVATE -O3 -Wall -Wextra -Wpedantic -Werror)
endif()
