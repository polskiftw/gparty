cmake_minimum_required(VERSION 3.28)
if(POLICY CMP0091)
  cmake_policy(SET CMP0091 NEW)
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(gdupe_codec_bench LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
find_package(JPEG REQUIRED)
find_package(PNG REQUIRED)
add_executable(gdupe_codec_bench ../benchmark_image_codecs.cpp)
target_link_libraries(gdupe_codec_bench PRIVATE JPEG::JPEG PNG::PNG windowscodecs ole32)
if(MSVC)
  target_compile_options(gdupe_codec_bench PRIVATE /O2 /W4 /permissive- /EHsc /utf-8)
  target_link_options(gdupe_codec_bench PRIVATE "$<$<CONFIG:Release>:/NODEFAULTLIB:LIBCMTD>")
endif()
